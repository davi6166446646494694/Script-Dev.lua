-- =====================================
-- ✨ INTRO NEON ANIMATION (DBU STYLE)
-- =====================================

local Players = game:GetService("Players")
local TweenService = game:GetService("TweenService")
local lp = Players.LocalPlayer

-- GUI
local gui = Instance.new("ScreenGui")
gui.Name = "NeonIntro"
gui.IgnoreGuiInset = true
gui.ResetOnSpawn = false
gui.Parent = lp:WaitForChild("PlayerGui")

-- FUNDO ESCURO
local bg = Instance.new("Frame", gui)
bg.Size = UDim2.new(1,0,1,0)
bg.BackgroundColor3 = Color3.fromRGB(5,5,5)
bg.BackgroundTransparency = 1

TweenService:Create(bg, TweenInfo.new(0.6), {
    BackgroundTransparency = 0
}):Play()

-- PAINEL CENTRAL
local panel = Instance.new("Frame", bg)
panel.Size = UDim2.new(0, 0, 0, 0)
panel.Position = UDim2.new(0.5,0,0.5,0)
panel.AnchorPoint = Vector2.new(0.5,0.5)
panel.BackgroundColor3 = Color3.fromRGB(0,0,0)
panel.BorderSizePixel = 0

-- CANTOS
Instance.new("UICorner", panel).CornerRadius = UDim.new(0,20)

-- NEON STROKE
local stroke = Instance.new("UIStroke", panel)
stroke.Thickness = 3

-- RGB LOOP
task.spawn(function()
    local h = 0
    while panel.Parent do
        h = (h + 1) % 360
        stroke.Color = Color3.fromHSV(h/360,1,1)
        task.wait(0.02)
    end
end)

-- ENTRADA DO PAINEL
TweenService:Create(panel, TweenInfo.new(0.8, Enum.EasingStyle.Back), {
    Size = UDim2.new(0, 420, 0, 220)
}):Play()

-- TÍTULO
local title = Instance.new("TextLabel", panel)
title.Size = UDim2.new(1,0,0,60)
title.BackgroundTransparency = 1
title.Text = "⚡ DBU SYSTEM ⚡"
title.TextColor3 = Color3.fromRGB(0,255,200)
title.Font = Enum.Font.Code
title.TextScaled = true
title.TextTransparency = 1

TweenService:Create(title, TweenInfo.new(0.6), {
    TextTransparency = 0
}):Play()

-- SUBTEXTO
local sub = Instance.new("TextLabel", panel)
sub.Position = UDim2.new(0,0,0,70)
sub.Size = UDim2.new(1,0,0,40)
sub.BackgroundTransparency = 1
sub.Text = "Inicializando módulos..."
sub.TextColor3 = Color3.fromRGB(200,200,200)
sub.Font = Enum.Font.Code
sub.TextScaled = true
sub.TextTransparency = 1

task.wait(0.3)
TweenService:Create(sub, TweenInfo.new(0.6), {
    TextTransparency = 0
}):Play()

-- LOADING BAR
local barBg = Instance.new("Frame", panel)
barBg.Position = UDim2.new(0.1,0,0.75,0)
barBg.Size = UDim2.new(0.8,0,0,12)
barBg.BackgroundColor3 = Color3.fromRGB(30,30,30)
Instance.new("UICorner", barBg).CornerRadius = UDim.new(1,0)

local bar = Instance.new("Frame", barBg)
bar.Size = UDim2.new(0,0,1,0)
bar.BackgroundColor3 = Color3.fromRGB(0,255,200)
Instance.new("UICorner", bar).CornerRadius = UDim.new(1,0)

-- ANIMAÇÃO DA BARRA
TweenService:Create(bar, TweenInfo.new(2), {
    Size = UDim2.new(1,0,1,0)
}):Play()

-- TEXTO DINÂMICO
task.spawn(function()
    local textos = {
        "Carregando scripts...",
        "Verificando permissões...",
        "Ativando sistemas...",
        "Pronto!"
    }

    for _, t in ipairs(textos) do
        sub.Text = t
        task.wait(0.6)
    end
end)

-- SAÍDA CINEMATOGRÁFICA
task.wait(3)

TweenService:Create(panel, TweenInfo.new(0.6, Enum.EasingStyle.Quad), {
    Size = UDim2.new(0,0,0,0)
}):Play()

TweenService:Create(bg, TweenInfo.new(0.6), {
    BackgroundTransparency = 1
}):Play()

task.wait(0.7)
gui:Destroy()
